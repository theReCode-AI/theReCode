gcloud artifacts repositories create harpic-cursor-v1  --repository-format=docker --location=europe-north1 --description="FAST API APP TEST" --immutable-tags --async

# ---------- Backend ----------
# Build context must be ./backend so Dockerfile copies backend/app/.env via COPY app
gcloud builds submit ./backend \
  --config=./backend/cloudbuild.yaml \
  --substitutions=_IMAGE=europe-north1-docker.pkg.dev/todo-app-506706/harpic-cursor-v1/harpic-cursor-v1-img:v1tag

# Cloud Run: localhost MongoDB in app/.env will NOT work.
# 1) Use MongoDB Atlas (mongodb+srv://...)
# 2) Atlas → Network Access → Add IP Address → Allow Access from Anywhere (0.0.0.0/0)
#    Cloud Run egress IPs are dynamic; without this you get TLS handshake errors.
# 3) Set env on deploy:
#
# gcloud run services update harpic-cursor-v1 \
#   --region europe-north1 \
#   --set-env-vars "THERECODE_MONGODB_URI=mongodb+srv://USER:PASS@CLUSTER/therecode?retryWrites=true&w=majority,THERECODE_WORKSPACE_ROOT=/workspace,THERECODE_ENVIRONMENT=production"

# Image is static nginx only (no proxy to "backend" hostname — that crashes nginx on Cloud Run).
